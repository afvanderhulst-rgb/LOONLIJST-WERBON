// =====================================================================
// MIT Werkbon - Firestore-regels
// Plakken in: Firebase console > Build > Firestore Database > tabblad Regels (Rules)
// 1. Selecteer ALLES wat daar staat en verwijder het.
// 2. Plak de VOLLEDIGE inhoud van dit bestand (deze uitleg mag mee, het zijn opmerkingen).
// 3. Klik op Publiceren (Publish). Een rode foutmelding? Dan is er iets niet helemaal meegekopieerd.
// Doe dit na elke update waarin staat dat de regels veranderd zijn.
// De beheerdersadressen staan bij isBeheerder() en magAccount(); pas ze daar aan als er een beheerder bijkomt of weggaat.
// =====================================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // e-mailadres van de ingelogde gebruiker in kleine letters (v173: hoofdletters in het adres tellen niet)
    function em() {
      return request.auth.token.email.lower();
    }
    function isBeheerder() {
      return request.auth != null
        && request.auth.token.email.lower() in ['a.f.vanderhulst@gmail.com', 'arjan@mit-installatietechniek.com']
        && request.auth.token.email_verified == true;
    }
    function isMit() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && (em().matches('.*@mit-installatietechniek[.]com$')
            || exists(/databases/$(database)/documents/toegang/$(em())))
        && !exists(/databases/$(database)/documents/geblokkeerd/$(em()));
    }
    function geblokkeerd() {
      let pad = /databases/$(database)/documents/geblokkeerd/$(em());
      return request.auth != null && exists(pad) && get(pad).data.get('volledig', false) == true;
    }
    // v173: mag dit adres een account hebben? (MIT-adres, beheerder, of toegelaten via toegang/ of kantoor/)
    function magAccount() {
      return request.auth != null
        && (em().matches('.*@mit-installatietechniek[.]com$')
            || em() in ['a.f.vanderhulst@gmail.com', 'arjan@mit-installatietechniek.com']
            || exists(/databases/$(database)/documents/toegang/$(em()))
            || exists(/databases/$(database)/documents/kantoor/$(em())));
    }
    function isKantoor() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && exists(/databases/$(database)/documents/kantoor/$(em()))
        && !geblokkeerd();
    }
    // v212: grenzen aan wat kantoor in een klaargezette bon zet (tegen te grote of rare gegevens)
    function inboxOk() {
      let d = request.resource.data;
      return d.size() <= 30
        && (!('opm' in d) || (d.opm is string && d.opm.size() <= 500))
        && (!('bestand' in d) || (d.bestand is string && d.bestand.size() <= 200))
        && (!('parts' in d) || (d.parts is int && d.parts <= 60))
        && (!('g' in d) || (d.g is map && d.g.size() <= 20));
    }
    function klantOk() {
      let d = request.resource.data;
      return d.keys().hasOnly(['naam','adres','plaats','zoek','door','t'])
        && d.naam is string && d.naam.size() > 0 && d.naam.size() <= 200
        && d.adres is string && d.adres.size() <= 200
        && d.plaats is string && d.plaats.size() <= 200
        && d.door is string && d.door.lower() == em();
    }
    match /users/{uid}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == uid && magAccount() && !geblokkeerd();
      allow read: if isBeheerder();
    }
    match /monteurs/{uid} {
      allow read: if isKantoor() || isBeheerder();
      allow create, update: if isBeheerder()
        || (request.auth != null && request.auth.uid == uid && !geblokkeerd()
            && request.resource.data.keys().hasOnly(['email','naam','rooster'])
            && request.resource.data.email is string && request.resource.data.email.lower() == em()
            && (!('naam' in request.resource.data)
                || (request.resource.data.naam is string && request.resource.data.naam.size() <= 80))
            && (!('rooster' in request.resource.data)
                || (request.resource.data.rooster is string && request.resource.data.rooster.size() <= 80)));
      allow delete: if isBeheerder() || (request.auth != null && request.auth.uid == uid);
    }
    match /foutlog/{id} {
      allow create: if request.auth != null && !geblokkeerd()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['t','uid','email','v','dev','soort','msg','src','n'])
        && request.resource.data.msg is string && request.resource.data.msg.size() <= 500
        && request.resource.data.src is string && request.resource.data.src.size() <= 300;
      allow read, delete: if isBeheerder();
    }
    // verzonden mails (schrijft alleen de server); de beheerder mag ze lezen (beheer.html > Systeem > Verzonden mails)
    match /mailuit/{id} {
      allow read: if isBeheerder();
    }
    match /inboxlog/{id} {
      allow create: if (isKantoor() || isBeheerder())
        && request.resource.data.door is string && request.resource.data.door.lower() == em()
        && request.resource.data.keys().hasOnly(['t','door','actie','wb','klant','voor','naar']);
      allow read, delete: if isBeheerder();
    }
    match /kantoor/{email} {
      allow get: if request.auth != null && em() == email;
      allow read, write: if isBeheerder();
      // v185: kantoor mag zelf collega's met een MIT-adres toegang geven of intrekken (niet zichzelf intrekken)
      allow list: if isKantoor();
      allow create: if isKantoor() && email.matches('.*@mit-installatietechniek[.]com$')
        && request.resource.data.keys().hasOnly(['t','door'])
        && request.resource.data.door is string && request.resource.data.door.lower() == em();
      allow delete: if isKantoor() && email.matches('.*@mit-installatietechniek[.]com$') && email != em();
    }
    match /inbox/{id} {
      allow read: if isBeheerder() || isKantoor()
        || (request.auth != null && resource.data.voor == request.auth.uid && !geblokkeerd());
      allow create: if (isBeheerder() || isKantoor()) && inboxOk();
      allow delete: if isBeheerder() || isKantoor();
      allow update: if ((isBeheerder() || isKantoor()) && inboxOk())
        || (request.auth != null && resource.data.voor == request.auth.uid && !geblokkeerd()
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','opgehaald','verstuurd','gereed','repm'])
            && request.resource.data.status == 'opgehaald')
        // v230: de monteur gooit een overgenomen bon weg die nog niet verstuurd is: weer klaargezet
        || (request.auth != null && resource.data.voor == request.auth.uid && !geblokkeerd()
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','opgehaald'])
            && resource.data.status == 'opgehaald' && request.resource.data.status == 'klaar'
            && !(resource.data.get('verstuurd', 0) is number && resource.data.get('verstuurd', 0) > 0));
      match /parts/{p} {
        allow read: if isBeheerder() || isKantoor()
          || (request.auth != null && !geblokkeerd()
              && get(/databases/$(database)/documents/inbox/$(id)).data.voor == request.auth.uid);
        allow write: if (isBeheerder() || isKantoor())
          && (request.resource == null || (request.resource.data.d is string && request.resource.data.d.size() <= 400000));
      }
    }
    // agenda (v180): planning uit de agenda-PDF van kantoor; de monteur leest alleen zijn eigen regels
    match /agenda/{id} {
      allow read: if isBeheerder() || isKantoor()
        || (request.auth != null && resource.data.voor == request.auth.uid && !geblokkeerd());
      allow write: if (isBeheerder() || isKantoor())
        && (request.resource == null || ((!('oms' in request.resource.data) || request.resource.data.oms.size() <= 1000)
            && request.resource.data.size() <= 30));
      // v192: de monteur meldt terug: bezig / verstuurd / werk gereed / reparatiemelding
      allow update: if request.auth != null && resource.data.voor == request.auth.uid && !geblokkeerd()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bezig','verstuurd','gereed','repm']);
    }
    // v194: verstuurde werkbonnen (datum, klant, gereed) voor het weekrooster van kantoor; de monteur schrijft zijn eigen
    match /bonlog/{id} {
      allow read: if isBeheerder() || isKantoor();
      allow create, update: if request.auth != null && !geblokkeerd()
        && request.resource.data.voor == request.auth.uid && id.matches(request.auth.uid + '_.*')
        && request.resource.data.keys().hasOnly(['voor','wb','klant','adres','plaats','tel','pr','mis','hist','dagen','van','tot','verstuurd','gereed','repm','t'])
        && request.resource.data.dagen is list && request.resource.data.dagen.size() <= 14
        && (!('hist' in request.resource.data) || (request.resource.data.hist is list && request.resource.data.hist.size() <= 20));
      allow delete: if isBeheerder();
    }
    // v197: verstuurde werkbon-PDF's (schrijft alleen de server); kantoor en beheerder mogen ze openen
    match /bonpdf/{id} {
      allow read: if isBeheerder() || isKantoor();
      allow delete: if isBeheerder(); // v231: beheerder verwijdert werkbonnen (kantoor-app > Zoeken)
      match /parts/{p} {
        allow read: if isBeheerder() || isKantoor();
        allow delete: if isBeheerder();
      }
    }
    // v210: correctie gevraagd door kantoor; de monteur leest zijn eigen en meldt terug (gezien / opnieuw verstuurd)
    match /correctie/{id} {
      allow read: if isBeheerder() || isKantoor()
        || (request.auth != null && resource.data.voor == request.auth.uid && !geblokkeerd());
      allow create: if (isBeheerder() || isKantoor())
        && request.resource.data.keys().hasOnly(['voor','voorNaam','wb','bon','ib','klant','reden','opm','door','t','status'])
        && request.resource.data.door is string && request.resource.data.door.lower() == em()
        && request.resource.data.reden is list && request.resource.data.reden.size() <= 8
        && request.resource.data.opm is string && request.resource.data.opm.size() <= 500
        && request.resource.data.status == 'open';
      allow delete: if isBeheerder() || isKantoor();
      allow update: if isBeheerder() || isKantoor()
        || (request.auth != null && resource.data.voor == request.auth.uid && !geblokkeerd()
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','gezien','opnieuw'])
            && request.resource.data.status in ['open','opnieuw']);
    }
    // v212: logboek van beheer- en kantooracties (alleen toevoegen; alleen de beheerder leest)
    match /beheerlog/{id} {
      allow create: if (isBeheerder() || isKantoor())
        && request.resource.data.keys().hasOnly(['t','door','actie','wat'])
        && request.resource.data.door is string && request.resource.data.door.lower() == em()
        && request.resource.data.actie is string && request.resource.data.actie.size() <= 120
        && request.resource.data.wat is string && request.resource.data.wat.size() <= 500;
      allow read, delete: if isBeheerder();
    }
    // v211: aandachtspunten die kantoor als afgehandeld heeft gemarkeerd
    match /afgehandeld/{id} {
      allow read, write: if isBeheerder() || isKantoor();
    }
    match /agendaimport/{id} {
      allow read: if isBeheerder() || isKantoor();
      allow create: if (isBeheerder() || isKantoor())
        && request.resource.data.door is string && request.resource.data.door.lower() == em();
      allow delete: if isBeheerder();
    }
    match /klanten/{id} {
      allow read: if isMit() || isBeheerder();
      allow create, update: if (isMit() || isBeheerder()) && klantOk();
      allow delete: if isBeheerder();
    }
    match /rooster/{doc} {
      allow read: if isMit() || isBeheerder() || isKantoor();
      allow write: if isBeheerder() || isKantoor();
    }
    match /klantlog/{id} {
      allow create: if (isMit() || isBeheerder())
        && request.resource.data.door is string && request.resource.data.door.lower() == em()
        && request.resource.data.keys().hasOnly(['k','soort','naam','adres','plaats','oud','door','t']);
      allow read, delete: if isBeheerder();
    }
    match /toegang/{email} {
      allow get: if request.auth != null && em() == email;
      allow read, write: if isBeheerder();
    }
    match /instellingen/{doc} {
      allow read: if request.auth != null;
      allow write: if isBeheerder();
    }
    match /geblokkeerd/{email} {
      allow get: if request.auth != null && em() == email;
      allow read, write: if isBeheerder();
    }
  }
}
